cmake_minimum_required(VERSION 3.16)
project(lr_mavros_bridge)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(mavros REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(rclcpp_components REQUIRED)
find_package(nlohmann_json REQUIRED)
include_directories(include)
find_package(tf2_msgs REQUIRED)
find_package(pluginlib REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_eigen REQUIRED)
find_package(diagnostic_updater REQUIRED)
find_package(mavros_msgs REQUIRED)
find_package(libmavconn REQUIRED)
find_package(angles REQUIRED)
find_package(eigen3_cmake_module REQUIRED)
find_package(Eigen3 REQUIRED)
find_library(GEOGRAPHIC_LIBRARY NAMES GeographicLib Geographic REQUIRED)
add_library(lr_replay_time SHARED src/replay_time.cpp src/source_time.cpp src/future_path.cpp)
ament_target_dependencies(lr_replay_time mavros rclcpp std_msgs pluginlib tf2_ros tf2_eigen diagnostic_updater mavros_msgs libmavconn angles)
target_link_libraries(lr_replay_time Eigen3::Eigen)
target_link_libraries(lr_replay_time nlohmann_json::nlohmann_json)
add_library(lr_net_sensor_relay SHARED src/net_sensor_relay.cpp)
ament_target_dependencies(lr_net_sensor_relay rclcpp rclcpp_components std_msgs sensor_msgs)
target_link_libraries(lr_net_sensor_relay nlohmann_json::nlohmann_json)
rclcpp_components_register_nodes(lr_net_sensor_relay "lr_mavros_bridge::NetSensorRelay")
pluginlib_export_plugin_description_file(mavros plugins.xml)
add_library(lr_geoid SHARED src/geoid.cpp)
target_link_libraries(lr_geoid ${GEOGRAPHIC_LIBRARY})
install(TARGETS lr_replay_time lr_geoid lr_net_sensor_relay LIBRARY DESTINATION lib)
add_executable(verify_tf_owner src/verify_tf_owner.cpp)
ament_target_dependencies(verify_tf_owner rclcpp tf2_msgs)
install(TARGETS verify_tf_owner DESTINATION lib/${PROJECT_NAME})
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/mavros_pose scripts/replay_clock_bridge scripts/live_runtime scripts/mavlink_future_path scripts/mission_path DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(live_relay test/test_live_relay.cpp)
  ament_target_dependencies(live_relay rclcpp rclcpp_components std_msgs sensor_msgs)
  target_link_libraries(live_relay nlohmann_json::nlohmann_json)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(causal_pose test/test_estimator.py)
  ament_add_pytest_test(ros_adapter test/test_node.py)
  ament_add_pytest_test(live_pose test/test_live_pose.py)
  ament_add_pytest_test(source_guard test/test_source_guard.py)
  ament_add_pytest_test(future_path test/test_future_path.py)
  ament_add_pytest_test(prediction_guard test/test_prediction_guard.py)
  ament_add_pytest_test(live_runtime test/test_live_runtime.py)
  ament_add_pytest_test(mission_path test/test_mission.py)
  ament_add_pytest_test(native_pipeline test/test_native_pipeline.py TIMEOUT 90)
  ament_add_pytest_test(live_gateway test/test_live_gateway.py TIMEOUT 25)
  ament_add_pytest_test(replay_clock test/test_replay_clock.py)
  ament_add_pytest_test(replay_clock_bridge test/test_clock_bridge.py)
endif()
ament_package()
