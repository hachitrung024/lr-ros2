cmake_minimum_required(VERSION 3.16)
project(lr_mavros_bridge)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(mavros REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(pluginlib REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_eigen REQUIRED)
find_package(diagnostic_updater REQUIRED)
find_package(mavros_msgs REQUIRED)
find_package(libmavconn REQUIRED)
find_package(angles REQUIRED)
find_package(eigen3_cmake_module REQUIRED)
find_package(Eigen3 REQUIRED)
find_library(GEOGRAPHIC_LIBRARY NAMES GeographicLib Geographic REQUIRED)
add_library(lr_replay_time SHARED src/replay_time.cpp)
ament_target_dependencies(lr_replay_time mavros rclcpp std_msgs pluginlib tf2_ros tf2_eigen diagnostic_updater mavros_msgs libmavconn angles)
target_link_libraries(lr_replay_time Eigen3::Eigen)
pluginlib_export_plugin_description_file(mavros plugins.xml)
add_library(lr_geoid SHARED src/geoid.cpp)
target_link_libraries(lr_geoid ${GEOGRAPHIC_LIBRARY})
install(TARGETS lr_replay_time lr_geoid LIBRARY DESTINATION lib)
add_executable(verify_tf_owner src/verify_tf_owner.cpp)
ament_target_dependencies(verify_tf_owner rclcpp tf2_msgs)
install(TARGETS verify_tf_owner DESTINATION lib/${PROJECT_NAME})
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/mavros_pose DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(causal_pose test/test_estimator.py)
  ament_add_pytest_test(ros_adapter test/test_node.py)
endif()
ament_package()
